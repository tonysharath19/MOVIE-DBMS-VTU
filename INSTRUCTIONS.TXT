STEP 3 — Create the database tables and initial cinema screens
Now we'll create the database structure for CINÉVERSE in pgAdmin 4.

We'll keep the database organized but appropriate for a college mini-project. We'll use one main SQL script instead of manually creating every table through multiple forms.

By the end of this step, you will have 9 tables and 3 screens with 20 seats each.
Sub-step 3.1 — Open the Query Tool
1. Open pgAdmin 4.
2. Expand Servers → PostgreSQL → Databases.
3. Click cineverse_db to select it.
4. Right-click cineverse_db.
5. Click Query Tool.
A SQL editor should open. This is where we'll execute our database commands.
Sub-step 3.2 — Copy the SQL script
Copy the entire script below and paste it into the Query Tool.

This script creates the tables for accounts, movies, screens, seats, shows, bookings, and payments. It also inserts the three initial screens and their seating layouts.

-- ============================================
-- CINEVERSE MOVIE BOOKING SYSTEM
-- STEP 3: DATABASE TABLES AND INITIAL SCREENS
-- ============================================

-- 1. USERS: Customers and administrators
CREATE TABLE IF NOT EXISTS users (
    user_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    full_name VARCHAR(100),
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash TEXT NOT NULL,
    date_of_birth DATE,
    phone VARCHAR(20),
    role VARCHAR(20) NOT NULL DEFAULT 'customer'
        CHECK (role IN ('customer', 'admin')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 2. MOVIES: Movie information
CREATE TABLE IF NOT EXISTS movies (
    movie_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    poster_path TEXT,
    genre VARCHAR(100),
    language VARCHAR(50),
    cast_members TEXT,
    plot TEXT,
    imdb_rating NUMERIC(3,1)
        CHECK (imdb_rating BETWEEN 0 AND 10),
    release_date DATE,
    status VARCHAR(20) NOT NULL DEFAULT 'upcoming'
        CHECK (status IN ('now_showing', 'upcoming', 'inactive')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 3. SCREENS: Cinema screen configuration and pricing
CREATE TABLE IF NOT EXISTS screens (
    screen_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    screen_name VARCHAR(50) NOT NULL UNIQUE,
    capacity INTEGER NOT NULL CHECK (capacity > 0),
    silver_price NUMERIC(10,2) NOT NULL DEFAULT 99
        CHECK (silver_price >= 0),
    gold_price NUMERIC(10,2) NOT NULL DEFAULT 199
        CHECK (gold_price >= 0),
    platinum_price NUMERIC(10,2) NOT NULL DEFAULT 299
        CHECK (platinum_price >= 0),
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 4. SEATS: Physical seats belonging to each screen
CREATE TABLE IF NOT EXISTS seats (
    seat_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    screen_id INTEGER NOT NULL
        REFERENCES screens(screen_id) ON DELETE CASCADE,
    seat_number VARCHAR(10) NOT NULL,
    seat_class VARCHAR(20) NOT NULL
        CHECK (seat_class IN ('silver', 'gold', 'platinum')),
    UNIQUE (screen_id, seat_number)
);

-- 5. SHOWS: Movie, screen, date and time
CREATE TABLE IF NOT EXISTS shows (
    show_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    movie_id INTEGER NOT NULL REFERENCES movies(movie_id),
    screen_id INTEGER NOT NULL REFERENCES screens(screen_id),
    show_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'scheduled'
        CHECK (status IN ('scheduled', 'cancelled')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CHECK (end_time > start_time),
    UNIQUE (screen_id, show_date, start_time)
);

-- 6. SHOW_SEATS: Availability and price for each seat at each show
CREATE TABLE IF NOT EXISTS show_seats (
    show_seat_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    show_id INTEGER NOT NULL REFERENCES shows(show_id),
    seat_id INTEGER NOT NULL REFERENCES seats(seat_id),
    seat_status VARCHAR(20) NOT NULL DEFAULT 'available'
        CHECK (
            seat_status IN ('available', 'booked', 'cinema_reserved')
        ),
    price NUMERIC(10,2) NOT NULL CHECK (price >= 0),
    UNIQUE (show_id, seat_id)
);

-- 7. BOOKINGS: Customer and counter bookings
CREATE TABLE IF NOT EXISTS bookings (
    booking_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    booking_code VARCHAR(40) NOT NULL UNIQUE,
    show_id INTEGER NOT NULL REFERENCES shows(show_id),
    user_id INTEGER REFERENCES users(user_id),
    booking_type VARCHAR(20) NOT NULL DEFAULT 'online'
        CHECK (booking_type IN ('online', 'counter')),
    customer_name VARCHAR(100) NOT NULL,
    customer_phone VARCHAR(20),
    booking_status VARCHAR(20) NOT NULL DEFAULT 'pending'
        CHECK (
            booking_status IN ('pending', 'confirmed', 'failed', 'cancelled')
        ),
    total_amount NUMERIC(10,2) NOT NULL DEFAULT 0
        CHECK (total_amount >= 0),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 8. BOOKING_SEATS: Seats allocated to a booking
CREATE TABLE IF NOT EXISTS booking_seats (
    booking_seat_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    booking_id INTEGER NOT NULL
        REFERENCES bookings(booking_id),
    show_seat_id INTEGER NOT NULL
        REFERENCES show_seats(show_seat_id),
    seat_price NUMERIC(10,2) NOT NULL CHECK (seat_price >= 0),
    is_active BOOLEAN NOT NULL DEFAULT TRUE
);

-- Prevent the same show-seat from being assigned
-- to multiple active booking records.
CREATE UNIQUE INDEX IF NOT EXISTS
    uq_booking_seats_one_active_booking
ON booking_seats(show_seat_id)
WHERE is_active = TRUE;

-- 9. PAYMENTS: Simulated payment records
CREATE TABLE IF NOT EXISTS payments (
    payment_id INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    booking_id INTEGER NOT NULL REFERENCES bookings(booking_id),
    payment_method VARCHAR(20) NOT NULL
        CHECK (payment_method IN ('upi', 'card', 'counter')),
    payment_status VARCHAR(20) NOT NULL DEFAULT 'initiated'
        CHECK (payment_status IN ('initiated', 'successful', 'failed')),
    amount NUMERIC(10,2) NOT NULL CHECK (amount >= 0),
    transaction_reference VARCHAR(100) UNIQUE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    paid_at TIMESTAMPTZ
);

-- ============================================
-- INITIAL CINEMA: THREE SCREENS
-- ============================================

INSERT INTO screens (
    screen_name,
    capacity,
    silver_price,
    gold_price,
    platinum_price
)
VALUES
    ('Screen 1', 20, 99, 199, 299),
    ('Screen 2', 20, 99, 199, 299),
    ('Screen 3', 20, 99, 199, 299)
ON CONFLICT (screen_name) DO NOTHING;

-- ============================================
-- INITIAL SEATING LAYOUT FOR ALL THREE SCREENS
--
-- Row A: 5 Silver seats       - Rs. 99 each
-- Rows B and C: 10 Gold seats - Rs. 199 each
-- Row D: 5 Platinum seats     - Rs. 299 each
-- ============================================

INSERT INTO seats (screen_id, seat_number, seat_class)
SELECT s.screen_id, v.seat_number, v.seat_class
FROM screens AS s
CROSS JOIN (
    VALUES
        ('A1', 'silver'),
        ('A2', 'silver'),
        ('A3', 'silver'),
        ('A4', 'silver'),
        ('A5', 'silver'),

        ('B1', 'gold'),
        ('B2', 'gold'),
        ('B3', 'gold'),
        ('B4', 'gold'),
        ('B5', 'gold'),

        ('C1', 'gold'),
        ('C2', 'gold'),
        ('C3', 'gold'),
        ('C4', 'gold'),
        ('C5', 'gold'),

        ('D1', 'platinum'),
        ('D2', 'platinum'),
        ('D3', 'platinum'),
        ('D4', 'platinum'),
        ('D5', 'platinum')
) AS v(seat_number, seat_class)
WHERE s.screen_name IN ('Screen 1', 'Screen 2', 'Screen 3')
ON CONFLICT (screen_id, seat_number) DO NOTHING;

Sub-step 3.5 — Confirm the three screens and their seats
In the Query Tool, run this verification query:
SELECT
    sc.screen_name,
    sc.capacity,
    COUNT(se.seat_id) AS actual_seat_count,
    COUNT(se.seat_id) FILTER (
        WHERE se.seat_class = 'silver'
    ) AS silver_seats,
    COUNT(se.seat_id) FILTER (
        WHERE se.seat_class = 'gold'
    ) AS gold_seats,
    COUNT(se.seat_id) FILTER (
        WHERE se.seat_class = 'platinum'
    ) AS platinum_seats
FROM screens AS sc
LEFT JOIN seats AS se ON se.screen_id = sc.screen_id
GROUP BY sc.screen_id, sc.screen_name, sc.capacity
ORDER BY sc.screen_id;


The expected results are:
Screen	Capacity	Actual seats	Silver	Gold	Platinum
Screen 1	20	20	5	10	5
Screen 2	20	20	5	10	5
Screen 3	20	20	5	10	5

Your initial seating configuration is now in the database. We haven't created movies, scheduled shows, or customer bookings yet, so those tables will remain empty.

One useful detail: The database stores individual seats separately from show-specific seat availability. That way, booking A1 for one show won't automatically book A1 for every other show.

const express = require("express");
const { Pool } = require("pg");
const bcrypt = require("bcryptjs");
require("dotenv").config();

const app = express();
const PORT = process.env.PORT || 3000;

// PostgreSQL database connection
const pool = new Pool({
    host: process.env.DB_HOST || "localhost",
    port: Number(process.env.DB_PORT || 5432),
    database: process.env.DB_NAME || "cineverse_db",
    user: process.env.DB_USER || "postgres",
    password: process.env.DB_PASSWORD
});

// Parse JSON requests
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Test/home page
app.get("/", (req, res) => {
    res.send(`
        <!DOCTYPE html>
        <html lang="en">
        <head>
            <meta charset="UTF-8">
            <meta name="viewport" content="width=device-width, initial-scale=1.0">
            <title>CINEVERSE</title>
        </head>
        <body>
            <h1>CINEVERSE</h1>
            <p>Your movie booking project server is running.</p>
            <a href="/api/db-test">Test database connection</a>
        </body>
        </html>
    `);
});

// Database connection test
app.get("/api/db-test", async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT
                current_database() AS database_name,
                current_user AS database_user
        `);

        res.json({
            success: true,
            message: "PostgreSQL connection successful!",
            database: result.rows[0].database_name,
            user: result.rows[0].database_user
        });
    } catch (error) {
        console.error("Database error:", error.message);

        res.status(500).json({
            success: false,
            message: "Database connection failed."
        });
    }
});

// Create the initial administrator if one does not exist
async function ensureAdminAccount() {
    const name = process.env.ADMIN_NAME;
    const email = process.env.ADMIN_EMAIL?.trim().toLowerCase();
    const password = process.env.ADMIN_PASSWORD;

    if (!name || !email || !password) {
        throw new Error(
            "ADMIN_NAME, ADMIN_EMAIL and ADMIN_PASSWORD must be set in .env"
        );
    }

    const existing = await pool.query(
        "SELECT user_id, role FROM users WHERE email = $1",
        [email]
    );

    if (existing.rows.length > 0) {
        if (existing.rows[0].role !== "admin") {
            throw new Error(
                "The configured admin email belongs to a non-admin account."
            );
        }

        console.log("Admin account already exists.");
        return;
    }

    // Hash the password before inserting it into the database.
    const passwordHash = await bcrypt.hash(password, 12);

    await pool.query(
        `INSERT INTO users
            (full_name, email, password_hash, role)
         VALUES ($1, $2, $3, 'admin')`,
        [name, email, passwordHash]
    );

    console.log("Initial admin account created successfully.");
}

// Start only after checking the database and admin setup
async function startServer() {
    try {
        await pool.query("SELECT 1");
        console.log("PostgreSQL connected successfully.");

        await ensureAdminAccount();

        app.listen(PORT, () => {
            console.log(
                `CINEVERSE running at http://localhost:${PORT}`
            );
        });
    } catch (error) {
        console.error("Startup failed:", error.message);
        await pool.end();
        process.exitCode = 1;
    }
}

startServer();



Save the file using Ctrl + S.
What this code does
1. Connects to your existing cineverse_db database.
2. Checks whether admin@movies.com exists in the users table.
3. If it doesn't exist, hashes Admin@123 and creates the administrator account.
4. If the admin already exists, it leaves the existing account and password unchanged.
5. Starts the server only after the database and admin setup succeed.

app.post("/api/register", async (req, res) => {
    try {
        const email = String(req.body.email || "")
            .trim()
            .toLowerCase();

        const password = req.body.password;

        if (
            !email ||
            typeof password !== "string" ||
            !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
        ) {
            return res.status(400).json({
                success: false,
                message: "Enter a valid email and password."
            });

        if (password.length < 8 || password.length > 72) {
            return res.status(400).json({
                success: false,
                message: "Password must contain 8 to 72 characters."
            });
        }

        const passwordHash = await bcrypt.hash(password, 12);

        const result = await pool.query(
            `INSERT INTO users (email, password_hash, role)
             VALUES ($1, $2, 'customer')
             RETURNING user_id, email, full_name, role,
                       date_of_birth, phone`,
            [email, passwordHash]
        );

        const user = result.rows[0];

        await startUserSession(req, user);

        res.status(201).json({
            success: true,
            message: "Account created. Complete your profile next.",
            requiresProfile: true,
            role: user.role
        });
    } catch (error) {
        if (error.code === "23505") {
            return res.status(409).json({
                success: false,
                message: "An account with this email already exists."
            });

        console.error("Registration failed:", error.message);

        res.status(500).json({
            success: false,
            message: "Registration failed. Please try again."
        });
    }
});

app.post("/api/login", async (req, res) => {
    try {
        const email = String(req.body.email || "")
            .trim()
            .toLowerCase();

        const password = req.body.password;

        if (!email || typeof password !== "string") {
            return res.status(400).json({
                success: false,
                message: "Email and password are required."
            });
        }

        const result = await pool.query(
            `SELECT user_id, email, password_hash, full_name,
                    role, date_of_birth, phone
             FROM users
             WHERE email = $1`,
            [email]
        );

        if (result.rows.length === 0) {
            return res.status(401).json({
                success: false,
                message: "Invalid email or password."
            });
        }

        const user = result.rows[0];

        const passwordMatches = await bcrypt.compare(
            password,
            user.password_hash
        );

        if (!passwordMatches) {
            return res.status(401).json({
                success: false,
                message: "Invalid email or password."
            });
        }

        await startUserSession(req, user);

        res.json({
            success: true,
            message: "Login successful.",
            role: user.role,
            requiresProfile:
                user.role === "customer" &&
                !isProfileComplete(user),
            user: {
                user_id: user.user_id,
                email: user.email,
                full_name: user.full_name,
                role: user.role
            }
        });
    } catch (error) {
        console.error("Login failed:", error.message);

        res.status(500).json({
            success: false,
            message: "Login failed. Please try again."
        });
    }
});

app.post("/api/profile", requireLogin, async (req, res) => {
    try {
        const user = req.session.user;

        if (user.role !== "customer") {
            return res.status(403).json({
                success: false,
                message: "This profile form is for customers."
            });
        }

        const fullName = String(req.body.fullName || "").trim();
        const dateOfBirth = String(req.body.dateOfBirth || "").trim();
        const phone = String(req.body.phone || "").trim();

        if (!fullName || fullName.length > 100) {
            return res.status(400).json({
                success: false,
                message: "Enter a name of up to 100 characters."
            });
        }

        if (!/^\d{4}-\d{2}-\d{2}$/.test(dateOfBirth)) {
            return res.status(400).json({
                success: false,
                message: "Enter a valid date of birth."
            });
        }

        const parsedDate = new Date(`${dateOfBirth}T00:00:00Z`);

        if (
            Number.isNaN(parsedDate.getTime()) ||
            parsedDate.toISOString().slice(0, 10) !== dateOfBirth ||
            dateOfBirth > new Date().toISOString().slice(0, 10)
        ) {
            return res.status(400).json({
                success: false,
                message: "Date of birth cannot be invalid or in the future."
            });
        }

        const phoneDigits = phone.replace(/\D/g, "");

        if (
            phoneDigits.length < 7 ||
            phoneDigits.length > 15 ||
            phone.length > 20
        ) {
            return res.status(400).json({
                success: false,
                message: "Enter a valid phone number."
            });
        }

        const result = await pool.query(
            `UPDATE users
             SET full_name = $1,
                 date_of_birth = $2,
                 phone = $3
             WHERE user_id = $4
             RETURNING user_id, email, full_name, role,
                       date_of_birth, phone`,
            [fullName, dateOfBirth, phone, user.user_id]
        );

        const updatedUser = result.rows[0];

        req.session.user = {
            ...req.session.user,
            full_name: updatedUser.full_name,
            date_of_birth: updatedUser.date_of_birth,
            phone: updatedUser.phone
        };

        res.json({
            success: true,
            message: "Profile saved successfully.",
            profileComplete: true,
            user: {
                user_id: updatedUser.user_id,
                email: updatedUser.email,
                full_name: updatedUser.full_name,
                role: updatedUser.role,
                date_of_birth: updatedUser.date_of_birth,
                phone: updatedUser.phone
            }
        });
    } catch (error) {
        console.error("Profile update failed:", error.message);

        res.status(500).json({
            success: false,
            message: "Unable to save profile."
        });
    }
});

async function ensureAdminAccount() {
    const name = process.env.ADMIN_NAME;
    const email = process.env.ADMIN_EMAIL?.trim().toLowerCase();
    const password = process.env.ADMIN_PASSWORD;

    if (!name || !email || !password) {
        throw new Error(
            "ADMIN_NAME, ADMIN_EMAIL and ADMIN_PASSWORD must be set in .env"
        );
    }

    if (password.length < 8 || password.length > 72) {
        throw new Error(
            "Admin password must contain 8 to 72 characters."
        );
    }

    const existing = await pool.query(
        "SELECT user_id, role FROM users WHERE email = $1",
        [email]
    );

    if (existing.rows.length > 0) {
        if (existing.rows[0].role !== "admin") {
            throw new Error(
                "The configured admin email belongs to a non-admin account."
            );
        }

        console.log("Admin account already exists.");
        return;
    }

    const passwordHash = await bcrypt.hash(password, 12);

    await pool.query(
        `INSERT INTO users
            (full_name, email, password_hash, role)
         VALUES ($1, $2, $3, 'admin')`,
        [name, email, passwordHash]
    );

    console.log("Initial admin account created successfully.");
}

async function startServer() {
    try {
        await pool.query("SELECT 1");
        console.log("PostgreSQL connected successfully.");

        await ensureAdminAccount();

        app.listen(PORT, () => {
            console.log(
                `CINEVERSE running at http://localhost:${PORT}`
            );
        });
    } catch (error) {
        console.error("Startup failed:", error.message);
        await pool.end();
        process.exitCode = 1;
    }
}

app.post("/api/login", async (req, res) => {
    try {
        const email = String(req.body.email || "")
            .trim()
            .toLowerCase();

        const password = req.body.password;

        if (!email || typeof password !== "string") {
            return res.status(400).json({
                success: false,
                message: "Email and password are required."
            });
        }

        const result = await pool.query(
            `SELECT user_id, email, password_hash, full_name,
                    role, date_of_birth, phone
             FROM users
             WHERE email = $1`,
            [email]
        );

        if (result.rows.length === 0) {
            return res.status(401).json({
                success: false,
                message: "Invalid email or password."
            });
        }

        const user = result.rows[0];

        const passwordMatches = await bcrypt.compare(
            password,
            user.password_hash
        );

        if (!passwordMatches) {
            return res.status(401).json({
                success: false,
                message: "Invalid email or password."
            });
        }

        await startUserSession(req, user);

        res.json({
            success: true,
            message: "Login successful.",
            role: user.role,
            requiresProfile:
                user.role === "customer" &&
                !isProfileComplete(user),
            user: {
                user_id: user.user_id,
                email: user.email,
                full_name: user.full_name,
                role: user.role
            }
        });
    } catch (error) {
        console.error("Login failed:", error.message);

        res.status(500).json({
            success: false,
            message: "Login failed. Please try again."
        });
    }
});

app.post("/api/profile", requireLogin, async (req, res) => {
    try {
        const user = req.session.user;

        if (user.role !== "customer") {
            return res.status(403).json({
                success: false,
                message: "This profile form is for customers."
            });
        }

        const fullName = String(req.body.fullName || "").trim();
        const dateOfBirth = String(req.body.dateOfBirth || "").trim();
        const phone = String(req.body.phone || "").trim();

        if (!fullName || fullName.length > 100) {
            return res.status(400).json({
                success: false,
                message: "Enter a name of up to 100 characters."
            });
        }

        if (!/^\d{4}-\d{2}-\d{2}$/.test(dateOfBirth)) {
            return res.status(400).json({
                success: false,
                message: "Enter a valid date of birth."
            });
        }

        const parsedDate = new Date(`${dateOfBirth}T00:00:00Z`);

        if (
            Number.isNaN(parsedDate.getTime()) ||
            parsedDate.toISOString().slice(0, 10) !== dateOfBirth ||
            dateOfBirth > new Date().toISOString().slice(0, 10)
        ) {
            return res.status(400).json({
                success: false,
                message: "Date of birth cannot be invalid or in the future."
            });
        }

        const phoneDigits = phone.replace(/\D/g, "");

        if (
            phoneDigits.length < 7 ||
            phoneDigits.length > 15 ||
            phone.length > 20
        ) {
            return res.status(400).json({
                success: false,
                message: "Enter a valid phone number."
            });
        }

        const result = await pool.query(
            `UPDATE users
             SET full_name = $1,
                 date_of_birth = $2,
                 phone = $3
             WHERE user_id = $4
             RETURNING user_id, email, full_name, role,
                       date_of_birth, phone`,
            [fullName, dateOfBirth, phone, user.user_id]
        );

        const updatedUser = result.rows[0];

        req.session.user = {
            ...req.session.user,
            full_name: updatedUser.full_name,
            date_of_birth: updatedUser.date_of_birth,
            phone: updatedUser.phone
        };

        res.json({
            success: true,
            message: "Profile saved successfully.",
            profileComplete: true,
            user: {
                user_id: updatedUser.user_id,
                email: updatedUser.email,
                full_name: updatedUser.full_name,
                role: updatedUser.role,
                date_of_birth: updatedUser.date_of_birth,
                phone: updatedUser.phone
            }
        });
    } catch (error) {
        console.error("Profile update failed:", error.message);

        res.status(500).json({
            success: false,
            message: "Unable to save profile."
        });
    }
});

// 11. INITIAL ADMIN ACCOUNT

async function ensureAdminAccount() {
    const name = process.env.ADMIN_NAME;
    const email = process.env.ADMIN_EMAIL?.trim().toLowerCase();
    const password = process.env.ADMIN_PASSWORD;

    if (!name || !email || !password) {
        throw new Error(
            "ADMIN_NAME, ADMIN_EMAIL and ADMIN_PASSWORD must be set in .env"
        );
    }

    if (password.length < 8 || password.length > 72) {
        throw new Error(
            "Admin password must contain 8 to 72 characters."
        );
    }

    const existing = await pool.query(
        "SELECT user_id, role FROM users WHERE email = $1",
        [email]
    );

    if (existing.rows.length > 0) {
        if (existing.rows[0].role !== "admin") {
            throw new Error(
                "The configured admin email belongs to a non-admin account."
            );
        }

        console.log("Admin account already exists.");
        return;
    }

    const passwordHash = await bcrypt.hash(password, 12);

    await pool.query(
        `INSERT INTO users
            (full_name, email, password_hash, role)
         VALUES ($1, $2, $3, 'admin')`,
        [name, email, passwordHash]
    );

    console.log("Initial admin account created successfully.");
}

async function startServer() {
    try {
        await pool.query("SELECT 1");
        console.log("PostgreSQL connected successfully.");

        await ensureAdminAccount();

        app.listen(PORT, () => {
            console.log(
                `CINEVERSE running at http://localhost:${PORT}`
            );
        });
    } catch (error) {
        console.error("Startup failed:", error.message);
        await pool.end();
        process.exitCode = 1;
    }
}
